# <staged path> is relative to the staged tree (the tarball's top-level directory, and the
# packagers' target/release); <install destination> is relative to the install root, so
# `usr/` is /usr for the packages and the user's prefix (~/.local) for install-user.sh.

# when the tables and this file differ.
